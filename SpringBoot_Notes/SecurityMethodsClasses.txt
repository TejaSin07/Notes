// ==============================
// SPRING SECURITY + JWT NOTES
// ==============================


// =========================================
// 1. OVERALL FLOW (VERY IMPORTANT)
// =========================================

// LOGIN FLOW:
// CustomerRestController.loginCheck() 
//  → AuthenticationManager.authenticate()   // entry point for authentication
//      → DaoAuthenticationProvider          // performs actual auth logic
//          → MyUserDetailsService.loadUserByUsername() // fetch user from DB
//          → PasswordEncoder.matches()      // compares raw vs encoded password
//  → JwtService.generateToken()             // create JWT
//  → return token to client


// REQUEST FLOW (AFTER LOGIN):
// Client sends JWT in header
//  → AppFilter.doFilterInternal()           // runs for every request
//      → extract token from header
//      → JwtService.extractUsername()       // get username from token
//      → MyUserDetailsService.loadUserByUsername() // load user
//      → JwtService.validateToken()         // validate token
//      → SecurityContextHolder.setAuthentication() // mark request authenticated
//  → Controller (API accessed)


// =========================================
// 2. IMPORTANT CLASSES (WITH PURPOSE)
// =========================================

// AppSecurityConfig
// → Central configuration class for Spring Security
// → Defines authentication rules, filter chain, and providers

// CustomerRestController
// → Entry point for login & register APIs
// → Starts authentication process

// AuthenticationManager
// → Main interface used to authenticate user
// → Delegates to AuthenticationProvider

// DaoAuthenticationProvider
// → Default provider that validates user using DB
// → Calls UserDetailsService + PasswordEncoder

// MyUserDetailsService
// → Custom service to fetch user from DB
// → Converts DB entity → Spring Security User

// PasswordEncoder (BCrypt)
// → Encodes password during registration
// → Matches password during login

// JwtService
// → Handles JWT creation, parsing, validation

// AppFilter (OncePerRequestFilter)
// → Runs on every request
// → Extracts and validates JWT

// SecurityContextHolder
// → Stores authentication for current request
// → If set → user is authenticated


// =========================================
// 3. IMPORTANT METHODS (WITH PURPOSE)
// =========================================

// AppSecurityConfig methods

securityFilterChain(HttpSecurity http)
// → Configures security rules (who can access what)

authenticationProvider()
// → Defines how authentication is performed

authenticationManager()
// → Provides AuthenticationManager bean

pwdEncoder()
// → Provides BCryptPasswordEncoder bean


// Controller method

loginCheck()
// → Accepts username/password
// → Calls AuthenticationManager
// → Generates JWT on success


// UserDetailsService method

loadUserByUsername(String username)
// → Fetch user from DB
// → Called automatically during authentication


// JWT Service methods

generateToken(String username)
// → Creates JWT with username + expiry

validateToken(String token, UserDetails userDetails)
// → Checks username + expiration

extractUsername(String token)
// → Extracts subject (username) from token


// Filter method

doFilterInternal(HttpServletRequest request, ...)
// → Intercepts every request
// → Validates JWT and sets authentication


// =========================================
// 4. LOGIN FLOW (STEP-BY-STEP)
// =========================================

// 1. User sends username + password

// 2. Controller creates authentication object
new UsernamePasswordAuthenticationToken(username, password)
// → Represents login request

// 3. AuthenticationManager.authenticate()
// → Triggers authentication process

// 4. Internally:
//    → loadUserByUsername()  // fetch user from DB
//    → passwordEncoder.matches(raw, encoded) // compare password

// 5. If success:
//    → JwtService.generateToken()
//    → return token


// =========================================
// 5. JWT STRUCTURE
// =========================================

// JWT contains:
// - Subject (username)
// - Issued time
// - Expiry time
// - Signature

// Example:
.setSubject(username)        // stores username
.setIssuedAt(new Date())     // token creation time
.setExpiration(...)          // expiry time
.signWith(key, HS256)        // digital signature


// =========================================
// 6. REQUEST FLOW (JWT VALIDATION)
// =========================================

// 1. Client sends request with header:
//    Authorization: Bearer <token>

// 2. Filter runs:

String authHeader = request.getHeader("Authorization");
// → fetch Authorization header

// 3. Extract token
token = authHeader.substring(7);
// → remove "Bearer "

// 4. Extract username
jwtService.extractUsername(token);

// 5. Load user from DB
userDetailsService.loadUserByUsername(username);

// 6. Validate token
jwtService.validateToken(token, userDetails);

// 7. Set authentication
SecurityContextHolder.getContext().setAuthentication(authToken);
// → marks request as authenticated


// =========================================
// 7. SECURITY CONFIG
// =========================================

// Disable CSRF (not needed for REST APIs)
http.csrf().disable()

// Public APIs
.requestMatchers("/api/login", "/api/register").permitAll()

// Secured APIs
.requestMatchers("/api/**").authenticated()

// Stateless session (no session, only JWT)
.sessionManagement()
.sessionCreationPolicy(SessionCreationPolicy.STATELESS)

// Add JWT filter before default auth filter
.addFilterBefore(filter, UsernamePasswordAuthenticationFilter.class)


// =========================================
// 8. PASSWORD HANDLING
// =========================================

// Register:
encodedPwd = passwordEncoder.encode(rawPwd)
// → store hashed password

// Login (internal):
passwordEncoder.matches(rawPwd, encodedPwd)
// → verify password


// =========================================
// 9. COMMON ISSUES
// =========================================

// 1. User not found → loadUserByUsername() fails
// 2. Password mismatch → matches() returns false
// 3. Token expired → validateToken() fails
// 4. Missing "Bearer " → token extraction fails
// 5. No roles → authorization not possible
// 6. SecurityContext not set → request blocked
